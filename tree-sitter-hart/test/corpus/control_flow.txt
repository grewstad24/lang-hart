func main() -> :none {
    mut arr := [1, 2, 3]

    repeat for arr using elem {
        print(elem)
    }

    repeat 10 using i {
        print(i)
    }

    repeat while x < 10 {
        continue
    }

    repeat until x >= 10 {
        break
    }

    match x using z {
        s32 => print("int")
        else => print("other")
    }

    switch y {
        1 => print("one")
        else => print("other")
    }
}
