fn main() -> :none {
    let arr := [1, 2, 3]

    for arr using elem {
        print(elem)
    }

    repeat 10 using i {
        print(i)
    }

    match x using z {
        s32 => print("int")
        else => print("other")
    }

    switch y {
        1 => print("one")
        else => print("other")
    }
}
